index: 0,

answers: [],

result: null,

};

/* ---------- 起動 ---------- */

init();

async function init() {

const params = new URLSearchParams(location.search);

state.source = params.get('src') || '';

state.demo = !CONFIG.LIFF_ID || !CONFIG.GAS_URL || params.get('demo') === '1' || location.protocol === 'file:';

try {

if (state.demo) {

const banner = document.getElementById('demo-banner');

banner.hidden = false;

if (location.protocol === 'file:' && CONFIG.LIFF_ID) banner.textContent = 'パソコンで直接開いたためデモ表示中（LINE連携はSTEP4の公開後に動きます）';

state.cfg = SAMPLE;

} else {

await loadScript('https://static.line-scdn.net/liff/edge/2/sdk.js');

await liff.init({ liffId: CONFIG.LIFF_ID });

if (!liff.isLoggedIn()) { liff.login({ redirectUri: location.href }); return; }

state.liffReady = true;

state.inLine = liff.isInClient();

try { state.profile = await liff.getProfile(); } catch (e) { /* 取得できなくても診断は続行 */ }

state.source = new URLSearchParams(location.search).get('src') || state.source;

const res = await fetch(CONFIG.GAS_URL + '?action=config');

state.cfg = await res.json();

if (!state.cfg || !state.cfg.ok) throw new Error('設定の読み込みに失敗しました');

}

renderStart();

} catch (err) {

document.getElementById('error-detail').textContent = String(err && err.message || err);

show('screen-error');

}

}

function loadScript(src) {

return new Promise((resolve, reject) => {

const s = document.createElement('script');

s.src = src; s.onload = resolve; s.onerror = () => reject(new Error('LINEの読み込みに失敗しました'));

document.head.appendChild(s);

});

}

/* ---------- 画面 ---------- */

function show(id) {

document.querySelectorAll('.screen').forEach(el => el.classList.toggle('active', el.id === id));

window.scrollTo(0, 0);

}

function setting(key, fallback) {

const v = state.cfg.settings && state.cfg.settings[key];

return (v === undefined || v === null || v === '') ? (fallback || '') : String(v);

}

function renderStart() {

document.getElementById('start-title').textContent = setting('diagnosis_title', '適職タイプ診断');

document.getElementById('start-subtitle').textContent = setting('diagnosis_subtitle');

document.getElementById('start-lead').textContent = setting('start_lead');

document.getElementById('start-note').textContent = setting('intro_note');

document.getElementById('btn-start').onclick = () => { state.index = 0; state.answers = []; renderQuestion(); };

show('screen-start');

}

function renderQuestion() {

const qs = state.cfg.questions;

const q = qs[state.index];

document.getElementById('progress-bar').style.width = (state.index / qs.length * 100) + '%';

document.getElementById('q-counter').textContent = 'Q' + (state.index + 1) + ' / ' + qs.length;

document.getElementById('q-text').textContent = q.text;

const box = document.getElementById('q-options');

box.innerHTML = '';

const options = q.kind === 'type' ? shuffle(q.options.slice()) : q.options;

options.forEach(opt => {

const b = document.createElement('button');

b.type = 'button';

b.className = 'option';

b.textContent = opt.label;

b.onclick = () => {

b.classList.add('picked');

state.answers[state.index] = opt;

setTimeout(next, 180);

};

box.appendChild(b);

});

document.getElementById('btn-back').style.visibility = state.index === 0 ? 'hidden' : 'visible';

document.getElementById('btn-back').onclick = () => { state.index--; renderQuestion(); };

show('screen-question');

}

function next() {

if (state.index < state.cfg.questions.length - 1) {

state.index++;

renderQuestion();

} else {

document.getElementById('progress-bar').style.width = '100%';

show('screen-judging');

setTimeout(finish, 1200);

}

}

/* ---------- 判定 ---------- */

function judge() {

const qs = state.cfg.questions;

const counts = {};

state.cfg.types.forEach(t => { counts[t.code] = 0; });

let tieBreak = '', motive = '', status = '';

qs.forEach((q, i) => {

const a = state.answers[i];

if (!a) return;

if (q.kind === 'type') {

counts[a.value] = (counts[a.value] || 0) + 1;

if (q.no === 'Q6') tieBreak = a.value;

} else if (q.kind === 'motive') {

motive = a.value;

} else if (q.kind === 'status') {

status = a.label;

}

});

const max = Math.max.apply(null, Object.values(counts));

const top = state.cfg.types.filter(t => counts[t.code] === max);

const type = top.find(t => t.code === tieBreak) || top[0];

const motiveObj = state.cfg.motives.find(m => m.code === motive) || null;

return { type, motive: motiveObj, statusLabel: status, counts };

}

async function finish() {

state.result = judge();

renderResult();

log('診断完了');

if (state.inLine && setting('send_result_to_chat').toUpperCase() === 'TRUE' && state.result.type.lineText) {

try { await liff.sendMessages([{ type: 'text', text: state.result.type.lineText }]); } catch (e) { /* 送信できなくても結果表示は続ける */ }

}

}

/* ---------- ボタンの動作 ---------- */

async function onInterview() {

log('面談ボタン');

const url = setting('interview_url');

if (isUrl(url)) { openUrl(url); return; }

if (state.inLine) {

try {

await liff.sendMessages([{ type: 'text', text: setting('interview_message', 'カジュアル面談を希望します') }]);

toast('トークに送信しました。担当者からご連絡します');

setTimeout(() => liff.closeWindow(), 1200);

} catch (e) { toast('送信できませんでした。トーク画面からご連絡ください'); }

} else {

toast(state.demo ? 'デモ：LINEアプリ内では「' + setting('interview_message', 'カジュアル面談を希望します') + '」がトークに送られます' : 'LINEアプリから開くと面談のお申し込みができます');

}

}

async function onShare() {

log('シェアボタン');

const typeName = state.result.type.name;

const link = CONFIG.LIFF_ID ? 'https://liff.line.me/' + CONFIG.LIFF_ID + '?src=share' : location.href;

const text = setting('share_text', 'Convolの適職タイプ診断、私は「{type}」だった！').replace('{type}', typeName) + '\n' + link;

try {

if (state.liffReady && liff.isApiAvailable('shareTargetPicker')) {

await liff.shareTargetPicker([{ type: 'text', text }]);

return;

}

if (navigator.share) { await navigator.share({ text }); return; }

} catch (e) { if (e && e.name === 'AbortError') return; /* 使えない環境ではコピーに切り替え */ }

await copyText(text);

toast('シェア用の文章をコピーしました');

}

/* ---------- 記録 ---------- */

function log(event) {

if (state.demo || !CONFIG.GAS_URL || !state.result) return;

const r = state.result;

const body = {

event,

userId: state.profile.userId || '',

displayName: state.profile.displayName || '',

typeName: r.type.name,

motiveLabel: r.motive ? r.motive.label : '',

statusLabel: r.statusLabel,

answers: state.cfg.questions.map((q, i) => q.no + ':' + (state.answers[i] ? state.answers[i].label : '')),

source: state.source,

};

fetch(CONFIG.GAS_URL, {

method: 'POST',

headers: { 'Content-Type': 'text/plain;charset=utf-8' },

body: JSON.stringify(body),

keepalive: true,

}).catch(() => {});

}

/* ---------- 小物 ---------- */

function el(tag, cls, text) {

const e = document.createElement(tag);

if (cls) e.className = cls;

if (text !== undefined) e.textContent = text;

return e;

}

function card(title) { const c = el('div', 'card'); c.append(el('h3', '', title)); return c; }

function isUrl(v) { return /^https?:\/\//i.test(String(v || '')); }

function openUrl(url) {

if (state.liffReady) liff.openWindow({ url, external: false });

else window.open(url, '_blank', 'noopener');

}

function shuffle(a) { for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [a[i], a[j]] = [a[j], a[i]]; } return a; }

async function copyText(text) {

try { await navigator.clipboard.writeText(text); return; } catch (e) { /* 下の方法で再試行 */ }

const t = document.createElement('textarea');

t.value = text; t.style.position = 'fixed'; t.style.opacity = '0';

document.body.append(t); t.select();

try { document.execCommand('copy'); } catch (e) { /* 何もしない */ }

t.remove();

}

let toastTimer;

function toast(msg) {

const t = document.getElementById('toast');

t.textContent = msg; t.classList.add('show');

clearTimeout(toastTimer); toastTimer = setTimeout(() => t.classList.remove('show'), 2600);

}
